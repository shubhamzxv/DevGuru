## Code example

Ten classics in compact form, run on sample inputs:

```python
from collections import Counter, deque
import heapq

def two_sum(nums, t):
    seen = {}
    for i, x in enumerate(nums):
        if t - x in seen: return [seen[t - x], i]
        seen[x] = i

def max_profit(prices):
    low, best = float("inf"), 0
    for p in prices:
        low, best = min(low, p), max(best, p - low)
    return best

def valid_parens(s):
    st, pair = [], {")": "(", "]": "[", "}": "{"}
    for c in s:
        if c in pair:
            if not st or st.pop() != pair[c]: return False
        else:
            st.append(c)
    return not st

def longest_unique(s):
    last, left, best = {}, 0, 0
    for r, c in enumerate(s):
        if last.get(c, -1) >= left: left = last[c] + 1
        last[c], best = r, max(best, r - left + 1)
    return best

def min_window(s, t):
    need, missing = Counter(t), len(t)
    left = start = 0
    end = float("inf")
    for right, c in enumerate(s, 1):
        if need[c] > 0: missing -= 1
        need[c] -= 1
        if missing == 0:
            while need[s[left]] < 0:
                need[s[left]] += 1
                left += 1
            if right - left < end - start: start, end = left, right
            need[s[left]] += 1
            missing += 1
            left += 1
    return "" if end == float("inf") else s[start:end]

def search_rotated(a, t):
    lo, hi = 0, len(a) - 1
    while lo <= hi:
        m = (lo + hi) // 2
        if a[m] == t: return m
        if a[lo] <= a[m]:
            lo, hi = (lo, m - 1) if a[lo] <= t < a[m] else (m + 1, hi)
        else:
            lo, hi = (m + 1, hi) if a[m] < t <= a[hi] else (lo, m - 1)
    return -1

def min_eating_speed(piles, h):
    lo, hi = 1, max(piles)
    while lo < hi:
        k = (lo + hi) // 2
        if sum((p + k - 1) // k for p in piles) <= h: hi = k
        else: lo = k + 1
    return lo

def can_finish(n, prereqs):
    graph, indeg = [[] for _ in range(n)], [0] * n
    for course, pre in prereqs:
        graph[pre].append(course); indeg[course] += 1
    q = deque(i for i in range(n) if indeg[i] == 0)
    done = 0
    while q:
        c = q.popleft(); done += 1
        for nxt in graph[c]:
            indeg[nxt] -= 1
            if indeg[nxt] == 0: q.append(nxt)
    return done == n

def coin_change(coins, amount):
    dp = [0] + [float("inf")] * amount
    for a in range(1, amount + 1):
        dp[a] = min((dp[a - c] + 1 for c in coins if c <= a), default=float("inf"))
    return dp[amount] if dp[amount] != float("inf") else -1

def top_k(nums, k):
    return [x for x, _ in heapq.nlargest(k, Counter(nums).items(), key=lambda p: p[1])]

print("two sum:", two_sum([3, 2, 4], 6))
print("stock:", max_profit([7, 1, 5, 3, 6, 4]))
print("parens:", valid_parens("()[]{}"), valid_parens("(]"))
print("unique substring:", longest_unique("abcabcbb"))
print("min window:", min_window("ADOBECODEBANC", "ABC"))
print("rotated:", search_rotated([4, 5, 6, 7, 0, 1, 2], 0))
print("koko speed:", min_eating_speed([3, 6, 7, 11], 8))
print("courses:", can_finish(2, [[1, 0]]), can_finish(2, [[1, 0], [0, 1]]))
print("coins:", coin_change([1, 2, 5], 11))
print("top k:", top_k([1, 1, 1, 2, 2, 3], 2))
```

<InterviewQA>

<QA q="Best time to buy and sell a stock once?">

Scan prices once keeping the lowest price seen so far and the best profit (current price − lowest so far). O(n) time, O(1) space.

</QA>

<QA q="Find the minimum window of s containing all characters of t.">

Sliding window with counts of what t needs and a counter of missing characters: expand right until nothing is missing, then shrink left while the window stays valid, recording the smallest window. O(|s| + |t|).

</QA>

<QA q="Can you finish all courses given prerequisites?">

Model courses as a directed graph and check for cycles: Kahn's topological sort counts processed courses starting from those with in-degree 0; if every course is processed there is no cycle and all can be finished. O(V + E).

</QA>

<QA q="How do you pick the minimum eating speed (Koko)?">

Binary search the speed k between 1 and the largest pile; for each k compute hours as the sum of ceil(pile / k) and check it is within h. The feasibility is monotonic, so find the smallest feasible k. O(n log max).

</QA>
